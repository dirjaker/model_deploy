# 本地模型部署 + 推理优化 — 技术设计文档

> 项目定位：一个完整的本地大模型部署方案，支持模型量化、推理加速、API 兼容、负载均衡，让企业可以在自有硬件上低成本、高性能地运行大模型。

## 一、项目背景与价值

### 1.1 为什么做这个项目？

很多企业有数据隐私要求（金融、医疗、政务），不能把数据发到外部 API。本地部署大模型是刚需，但面临几个核心挑战：

1. **显存不够**：70B 模型需要 140GB 显存（FP16），消费级显卡跑不动
2. **速度太慢**：原始 PyTorch 推理吞吐量低，无法支撑生产流量
3. **接口不兼容**：现有应用都是按 OpenAI API 格式开发的
4. **运维困难**：模型加载、显存管理、故障恢复都需要自动化

这个项目展示了：
- 对 **模型推理全链路** 的深入理解（量化 → 加速 → 服务 → 监控）
- 对 **GPU 编程和系统优化** 的实践能力
- 对 **生产级部署** 的工程化思维

### 1.2 面试话术

> "我做了一个本地大模型部署方案，通过 GPTQ/AWQ 量化把 70B 模型压缩到单卡可运行，用 vLLM 做推理加速实现了 10 倍吞吐提升，并且提供 OpenAI 兼容 API，支持负载均衡和自动扩缩容。"

## 三、核心模块设计

### 3.1 模型量化模块

#### 3.1.1 量化技术对比

| 量化方法 | 位数 | 精度损失 | 速度提升 | 显存节省 | 适用场景 |
|---------|------|---------|---------|---------|---------|
| **GPTQ** | 4-bit | 低 | 中 | 75% | GPU 推理，精度敏感 |
| **AWQ** | 4-bit | 最低 | 高 | 75% | GPU 推理，速度优先 |
| **GGUF** | 2-8 bit | 可调 | 中 | 可调 | CPU/混合推理 |
| **FP8** | 8-bit | 极低 | 高 | 50% | H100/A100 显卡 |
| **INT8** | 8-bit | 低 | 中 | 50% | 通用 GPU |
| **INT4** | 4-bit | 中 | 高 | 75% | 显存受限场景 |

#### 3.1.2 GPTQ 量化实现

```python
"""
GPTQ 量化流程：
1. 加载原始 FP16 模型
2. 准备校准数据集（128-256 条样本）
3. 逐层量化：找到最优量化参数，最小化输出误差
4. 保存量化模型
"""

import torch
from transformers import AutoModelForCausalLM, AutoTokenizer
from gptqmodel import GPTQ, load_quantized_model

class GPTQQuantizer:
    """GPTQ 量化器"""

    def __init__(
        self,
        model_path: str,
        bits: int = 4,
        group_size: int = 128,
        desc_act: bool = True,
        damp_percent: float = 0.01,
    ):
        self.model_path = model_path
        self.bits = bits
        self.group_size = group_size
        self.desc_act = desc_act       # 激活排序（提升精度）
        self.damp_percent = damp_percent

    def prepare_calibration_data(
        self,
        dataset_name: str = "wikitext",
        num_samples: int = 256,
        max_length: int = 2048,
    ) -> list[str]:
        """
        准备校准数据集

        校准数据的选择直接影响量化质量：
        - 数据量：128-256 条通常足够
        - 数据分布：应与实际使用场景接近
        - 长度：覆盖模型的最大上下文长度
        """
        from datasets import load_dataset

        dataset = load_dataset(dataset_name, split="train")
        samples = []
        for i, item in enumerate(dataset):
            if len(samples) >= num_samples:
                break
            text = item["text"]
            if len(text) > 100:  # 过滤太短的样本
                samples.append(text[:max_length])

        return samples

    async def quantize(self, output_path: str, calibration_data: list[str] = None):
        """
        执行量化

        核心算法（简化版）：
        1. 对每一层的权重矩阵 W
        2. 计算 H = X^T * X（Hessian 矩阵的近似）
        3. 对每一列：
           a. 找到最优量化参数（scale, zero_point）
           b. 量化权重
           c. 计算量化误差
           d. 将误差传播到后续列（OBQ 方法）
        4. 保存量化后的权重
        """
        if calibration_data is None:
            calibration_data = self.prepare_calibration_data()

        # 加载模型
        model = AutoModelForCausalLM.from_pretrained(
            self.model_path,
            torch_dtype=torch.float16,
            device_map="auto",
        )
        tokenizer = AutoTokenizer.from_pretrained(self.model_path)

        # 配置量化参数
        quantize_config = {
            "bits": self.bits,
            "group_size": self.group_size,
            "desc_act": self.desc_act,
            "damp_percent": self.damp_percent,
            "sym": False,               # 非对称量化（精度更高）
            "true_sequential": True,    # 逐层顺序量化
        }

        # 执行量化
        quantizer = GPTQ(model, tokenizer, quantize_config)
        quantizer.quantize(calibration_data)

        # 保存
        quantizer.save(output_path)
        tokenizer.save_pretrained(output_path)

        # 验证量化质量
        self._validate_quantization(output_path)

        return output_path

    def _validate_quantization(self, quantized_path: str):
        """验证量化模型的质量"""
        model = load_quantized_model(quantized_path)
        tokenizer = AutoTokenizer.from_pretrained(quantized_path)

        # 困惑度测试（Perplexity）
        test_data = self.prepare_calibration_data(num_samples=10)
        total_loss = 0
        total_tokens = 0

        for text in test_data:
            inputs = tokenizer(text, return_tensors="pt", truncation=True, max_length=512)
            with torch.no_grad():
                outputs = model(**inputs, labels=inputs["input_ids"])
            total_loss += outputs.loss.item() * inputs["input_ids"].numel()
            total_tokens += inputs["input_ids"].numel()

        avg_ppl = torch.exp(torch.tensor(total_loss / total_tokens)).item()
        print(f"量化模型困惑度: {avg_ppl:.2f}")

        # 基准：FP16 模型困惑度通常在 5-15，量化后不应超过 1.5 倍
        if avg_ppl > 30:
            print("⚠️ 警告：量化质量较差，建议增加校准数据或调整参数")


# ========== AWQ 量化（推荐） ==========

class AWQQuantizer:
    """
    AWQ (Activation-aware Weight Quantization) 量化器

    核心思想：不是均匀量化所有权重，而是保护"重要"通道
    - 重要通道 = 激活值大的通道
    - 对重要通道用更高精度或缩放因子保护
    - 实现：搜索每个通道的最优缩放因子 s，然后量化 W * s

    优势：
    - 比 GPTQ 精度更高（尤其在 4-bit 下）
    - 推理速度更快（支持 FP16 混合精度）
    - 对校准数据不敏感
    """

    def __init__(self, model_path: str, bits: int = 4, group_size: int = 128):
        self.model_path = model_path
        self.bits = bits
        self.group_size = group_size

    async def quantize(self, output_path: str):
        from awq import AutoAWQForCausalLM

        # 加载模型
        model = AutoAWQForCausalLM.from_pretrained(
            self.model_path,
            torch_dtype=torch.float16,
        )
        tokenizer = AutoTokenizer.from_pretrained(self.model_path)

        # 配置
        quant_config = {
            "zero_point": True,
            "q_group_size": self.group_size,
            "w_bit": self.bits,
            "version": "GEMM",  # GEMM kernel 更快
        }

        # 执行量化（自动搜索最优缩放因子）
        model.quantize(
            tokenizer,
            quant_config=quant_config,
            calib_data="wikitext",    # 校准数据集
            max_calib_samples=128,
            max_calib_seq_len=512,
        )

        # 保存
        model.save_quantized(output_path)
        tokenizer.save_pretrained(output_path)

        return output_path
```

```python
class GGUFConverter:
    """
    GGUF 量化转换器

    GGUF 是 llama.cpp 使用的格式，支持 CPU 推理：
    - Q4_K_M: 4-bit 量化，质量/大小平衡（推荐）
    - Q5_K_M: 5-bit 量化，质量更高
    - Q8_0:   8-bit 量化，接近 FP16 质量
    - Q2_K:   2-bit 量化，极度压缩
    """

    QUANT_TYPES = {
        "Q2_K":   {"bits": 2, "quality": "低", "size_ratio": "0.15x"},
        "Q3_K_M": {"bits": 3, "quality": "中低", "size_ratio": "0.20x"},
        "Q4_K_M": {"bits": 4, "quality": "中", "size_ratio": "0.25x"},
        "Q5_K_M": {"bits": 5, "quality": "中高", "size_ratio": "0.35x"},
        "Q6_K":   {"bits": 6, "quality": "高", "size_ratio": "0.40x"},
        "Q8_0":   {"bits": 8, "quality": "极高", "size_ratio": "0.50x"},
    }

    async def convert(
        self,
        model_path: str,
        output_path: str,
        quant_type: str = "Q4_K_M",
    ):
        """
        转换流程：
        1. HF 模型 → GGML 格式（转换权重格式）
        2. GGML → GGUF 格式（添加元数据）
        3. GGUF → 量化 GGUF（执行量化）
        """
        import subprocess

        # Step 1: 转换为 GGML
        subprocess.run([
            "python", "convert_hf_to_gguf.py",
            model_path,
            "--outfile", f"{output_path}/model.ggml",
            "--outtype", "f16",
        ], check=True)

        # Step 2: 量化
        subprocess.run([
            "./llama-quantize",
            f"{output_path}/model.ggml",
            f"{output_path}/model-{quant_type}.gguf",

            "./llama-perplexity",

            "-f", "wikitext-2-raw/wiki.test.raw",

        print(f"Perplexity: {result.stdout}")

        return f"{output_path}/model-{quant_type}.gguf"
```

#### 3.2.1 vLLM 部署方案

```python
"""
vLLM 核心优化技术：

1. PagedAttention
   - 传统方式：每个请求预分配最大长度的 KV Cache → 显存浪费
   - PagedAttention：按页管理 KV Cache（类似操作系统虚拟内存）
   - 效果：显存利用率提升 2-4 倍，可同时处理更多请求

2. Continuous Batching
   - 传统方式：等一个 batch 全部完成再处理下一个
   - Continuous：请求完成立即释放，新请求立即加入
   - 效果：吞吐量提升 2-3 倍

3. Speculative Decoding（投机采样）
   - 用小模型快速生成候选 token
   - 大模型并行验证多个 token
   - 效果：延迟降低 2-3 倍（不损失质量）

4. Tensor Parallelism
   - 将模型权重分布到多张 GPU
   - 支持单机多卡推理大模型
"""

from vllm import AsyncLLMEngine, AsyncEngineArgs, SamplingParams

class VLLMServer:
    """vLLM 推理服务器"""

    def __init__(self, config: dict):
        self.config = config

    async def create_engine(self):
        """创建推理引擎"""
        engine_args = AsyncEngineArgs(
            model=self.config["model_path"],
            # 显存管理
            gpu_memory_utilization=0.90,    # GPU 显存使用率（预留 10% 给系统）
            max_model_len=self.config.get("max_model_len", 8192),
            # 量化
            quantization=self.config.get("quantization"),  # "awq" | "gptq" | None
            # 并行
            tensor_parallel_size=self.config.get("tensor_parallel", 1),
            # 批处理
            max_num_batched_tokens=self.config.get("max_batched_tokens", 32768),
            max_num_seqs=self.config.get("max_num_seqs", 256),
            # 性能
            enforce_eager=self.config.get("enforce_eager", False),  # CUDA Graph
            enable_prefix_caching=True,     # 前缀缓存（相同前缀复用 KV Cache）
            # 数据类型
            dtype="auto",                   # 自动选择最优精度
        )

        self.engine = AsyncLLMEngine.from_engine_args(engine_args)
        return self.engine

    async def generate(
        self,
        prompt: str,
        max_tokens: int = 1024,
        temperature: float = 0.7,
        top_p: float = 0.9,
        stop: list[str] = None,
    ):
        """生成文本"""
        sampling_params = SamplingParams(
            max_tokens=max_tokens,
            temperature=temperature,
            top_p=top_p,
            stop=stop,
        )

        request_id = f"req-{uuid.uuid4()}"
        results = []

        async for output in self.engine.generate(
            prompt=prompt,
            sampling_params=sampling_params,
            request_id=request_id,
        ):
            results.append(output)

        final = results[-1]
        return {
            "text": final.outputs[0].text,
            "tokens": len(final.outputs[0].token_ids),
            "finish_reason": final.outputs[0].finish_reason,
        }
```

#### 3.2.2 llama.cpp 部署方案（CPU / 混合推理）

```python
"""
llama.cpp 适用场景：
- 没有 GPU 的服务器
- 需要在 CPU 上高效推理
- 边缘设备部署（树莓派、Mac Mini）
- 量化模型快速测试
"""

class LlamaCppServer:
    """llama.cpp 推理服务器"""

    def __init__(self, config: dict):
        self.config = config

    async def start_server(self):
        """启动 llama.cpp 服务器"""
        import subprocess

        cmd = [
            "./llama-server",
            "-m", self.config["model_path"],
            "-c", str(self.config.get("context_size", 4096)),
            "-t", str(self.config.get("threads", 8)),
            "-ngl", str(self.config.get("gpu_layers", 0)),  # GPU 卸载层数
            "--port", str(self.config.get("port", 8000)),
            "--host", "0.0.0.0",
        ]

        # Metal 加速（macOS）
        if self.config.get("metal"):
            cmd.append("--metal")

        # CUDA 加速（Linux + NVIDIA GPU）
        if self.config.get("cuda"):
            cmd.extend(["-ngl", "99"])  # 全部层卸载到 GPU

        process = subprocess.Popen(cmd)
        return process

    def optimize_for_hardware(self, hardware_info: dict) -> dict:
        """
        根据硬件自动选择最优配置

        策略：
        - 有 GPU → 尽量多层卸载到 GPU
        - CPU 核心多 → 增加线程数
        - 内存大 → 增加上下文长度
        """
        config = self.config.copy()

        if hardware_info.get("gpu_memory_gb", 0) >= 24:
            # 24GB 显存：70B Q4 模型可以部分卸载
            config["gpu_layers"] = 40
            config["context_size"] = 8192
        elif hardware_info.get("gpu_memory_gb", 0) >= 8:
            # 8GB 显存：7B-13B 模型可以全部卸载
            config["gpu_layers"] = 99
            config["context_size"] = 4096
        else:
            # 无 GPU：纯 CPU 推理
            config["gpu_layers"] = 0
            config["threads"] = min(hardware_info.get("cpu_cores", 4), 16)
            config["context_size"] = 2048

        return config
```

#### 3.2.3 性能基准测试

```python
import asyncio
import time
import statistics

class BenchmarkRunner:
    """推理性能基准测试"""

    async def run_benchmark(
        self,
        server_url: str,
        test_cases: list[dict],
        concurrency_levels: list[int] = [1, 2, 4, 8, 16, 32],
    ) -> dict:
        """
        基准测试指标：
        - TTFT (Time to First Token): 首 token 延迟
        - TPS (Tokens per Second): 吞吐量
        - P50/P95/P99 延迟
        - 并发处理能力
        """
        results = {}

        for concurrency in concurrency_levels:
            print(f"\n--- 并发数: {concurrency} ---")
            latencies = []
            ttft_list = []
            tps_list = []

            async def single_request(prompt: str):
                import httpx
                start = time.time()
                first_token_time = None
                tokens = 0

                async with httpx.AsyncClient() as client:
                    async with client.stream(
                        "POST",
                        f"{server_url}/v1/chat/completions",
                        json={
                            "model": "default",
                            "messages": [{"role": "user", "content": prompt}],
                            "stream": True,
                            "max_tokens": 256,
                        },
                        timeout=120,
                    ) as response:
                        async for line in response.aiter_lines():
                            if line.startswith("data: ") and line != "data: [DONE]":
                                if first_token_time is None:
                                    first_token_time = time.time()
                                tokens += 1

                end = time.time()
                total_time = end - start
                ttft = (first_token_time or end) - start

                latencies.append(total_time)
                ttft_list.append(ttft)
                if total_time > 0:
                    tps_list.append(tokens / total_time)

            # 并发执行
            tasks = []
            for i in range(concurrency):
                prompt = test_cases[i % len(test_cases)]["prompt"]
                tasks.append(single_request(prompt))

            await asyncio.gather(*tasks)

            results[concurrency] = {
                "avg_latency_ms": statistics.mean(latencies) * 1000,
                "p50_latency_ms": statistics.median(latencies) * 1000,
                "p95_latency_ms": sorted(latencies)[int(len(latencies) * 0.95)] * 1000,
                "p99_latency_ms": sorted(latencies)[int(len(latencies) * 0.99)] * 1000,
                "avg_ttft_ms": statistics.mean(ttft_list) * 1000,
                "avg_tps": statistics.mean(tps_list),
                "total_tokens": sum(tps_list) * statistics.mean(latencies),
            }

            print(f"  平均延迟: {results[concurrency]['avg_latency_ms']:.0f}ms")
            print(f"  P95 延迟: {results[concurrency]['p95_latency_ms']:.0f}ms")
            print(f"  平均 TTFT: {results[concurrency]['avg_ttft_ms']:.0f}ms")
            print(f"  平均 TPS: {results[concurrency]['avg_tps']:.1f}")

        return results

    def generate_report(self, results: dict) -> str:
        """生成性能报告"""
        report = ["# 推理性能基准测试报告\n"]
        report.append("| 并发数 | 平均延迟 | P95 延迟 | TTFT | TPS |")
        report.append("|--------|---------|---------|------|-----|")

        for concurrency, metrics in sorted(results.items()):
            report.append(
                f"| {concurrency} "
                f"| {metrics['avg_latency_ms']:.0f}ms "
                f"| {metrics['p95_latency_ms']:.0f}ms "
                f"| {metrics['avg_ttft_ms']:.0f}ms "
                f"| {metrics['avg_tps']:.1f} |"
            )

        return "\n".join(report)
```

```python
class LoadBalancer:
    """负载均衡器"""

    def __init__(self):
        self.backends: list[dict] = []  # [{"url": "...", "weight": 1, "healthy": True}]
        self._current_index = 0

    def add_backend(self, url: str, weight: int = 1):
        self.backends.append({
            "url": url,
            "weight": weight,
            "healthy": True,
            "active_requests": 0,
            "avg_latency_ms": 0,
        })

    def get_backend(self, strategy: str = "least_connections") -> str:
        """
        选择后端

        策略：
        1. round_robin: 轮询
        2. least_connections: 最少连接
        3. weighted: 加权轮询
        4. least_latency: 最低延迟
        """
        healthy = [b for b in self.backends if b["healthy"]]
        if not healthy:
            raise HTTPException(503, "所有后端不可用")

        if strategy == "round_robin":
            backend = healthy[self._current_index % len(healthy)]
            self._current_index += 1
        elif strategy == "least_connections":
            backend = min(healthy, key=lambda b: b["active_requests"])
        elif strategy == "least_latency":
            backend = min(healthy, key=lambda b: b["avg_latency_ms"])
        else:
            backend = healthy[0]

        return backend["url"]

    async def health_check(self):
        """定期健康检查"""
        while True:
            for backend in self.backends:
                try:
                    async with httpx.AsyncClient() as client:
                        resp = await client.get(f"{backend['url']}/health", timeout=5)
                        backend["healthy"] = resp.status_code == 200
                except:
                    backend["healthy"] = False
            await asyncio.sleep(10)
```

#### 3.3.2 语义缓存

```python
class SemanticCache:
    """
    语义缓存 — 相似请求复用结果

    原理：
    1. 用户提问 → Embedding → 向量检索
    2. 如果找到相似问题（余弦相似度 > 0.95）→ 直接返回缓存结果
    3. 未命中 → 正常推理 → 缓存结果

    适用场景：
    - 客服系统（相似问题多）
    - FAQ 问答
    - 测试/开发环境（重复请求多）
    """

    def __init__(self, embedding_model, vector_store, redis_client):
        self.embedding = embedding_model
        self.vector_store = vector_store
        self.redis = redis_client
        self.similarity_threshold = 0.95
        self.ttl = 3600 * 24  # 缓存 24 小时

    async def get(self, query: str) -> dict | None:
        """查询缓存"""
        # 1. 精确匹配（hash）
        query_hash = hashlib.md5(query.encode()).hexdigest()
        cached = await self.redis.get(f"cache:{query_hash}")
        if cached:
            return json.loads(cached)

        # 2. 语义匹配（向量检索）
        query_embedding = await self.embedding.encode(query)
        similar = await self.vector_store.search(
            query_embedding, top_k=1, threshold=self.similarity_threshold
        )

        if similar:
            cache_key = similar[0]["metadata"]["cache_key"]
            cached = await self.redis.get(f"cache:{cache_key}")
            if cached:
                return json.loads(cached)

        return None

    async def set(self, query: str, response: dict):
        """写入缓存"""
        # 1. 存入 Redis（精确匹配）
        query_hash = hashlib.md5(query.encode()).hexdigest()
        await self.redis.setex(
            f"cache:{query_hash}",
            self.ttl,
            json.dumps(response, ensure_ascii=False),
        )

        # 2. 存入向量库（语义匹配）
        query_embedding = await self.embedding.encode(query)
        await self.vector_store.insert(
            text=query,
            embedding=query_embedding,
            metadata={"cache_key": query_hash},
        )
```

```python
from pathlib import Path
import json

class ModelManager:
    """模型生命周期管理"""

    def __init__(self, models_dir: str = "/models"):
        self.models_dir = Path(models_dir)
        self.registry: dict[str, dict] = {}

    def register_model(
        self,
        name: str,
        path: str,
        quantization: str = None,
        format: str = "hf",  # hf | gguf | onnx
        description: str = "",
    ):
        """注册模型"""
        self.registry[name] = {
            "name": name,
            "path": path,
            "quantization": quantization,
            "format": format,
            "description": description,
            "size_gb": self._get_model_size(path),
            "vram_required_gb": self._estimate_vram(path, quantization),
            "registered_at": datetime.now().isoformat(),
        }

    def get_model_info(self, name: str) -> dict:
        """获取模型信息"""
        return self.registry.get(name)

    def recommend_model(
        self,
        gpu_memory_gb: float,
        task: str = "chat",
        quality_priority: str = "balanced",
    ) -> str:
        """
        根据硬件推荐模型

        策略：
        - 显存 >= 80GB → FP16 全精度
        - 显存 >= 24GB → 70B AWQ 4-bit 或 13B FP16
        - 显存 >= 12GB → 7B AWQ 4-bit 或 13B GGUF Q4
        - 显存 >= 6GB  → 7B GGUF Q4
        - 无 GPU      → 7B GGUF Q4 (CPU)
        """
        candidates = []
        for name, info in self.registry.items():
            if info["vram_required_gb"] <= gpu_memory_gb:
                candidates.append((name, info))

        if not candidates:
            return None

        # 按质量优先级排序
        if quality_priority == "quality":
            candidates.sort(key=lambda x: -x[1]["size_gb"])
        elif quality_priority == "speed":
            candidates.sort(key=lambda x: x[1]["size_gb"])
        else:
            # balanced: 选择刚好能装下的最大模型
            candidates.sort(key=lambda x: -x[1]["size_gb"])

        return candidates[0][0]

    def _get_model_size(self, path: str) -> float:
        """计算模型文件大小（GB）"""
        model_path = Path(path)
        if model_path.is_dir():
            total = sum(f.stat().st_size for f in model_path.rglob("*") if f.is_file())
        else:
            total = model_path.stat().st_size
        return total / (1024 ** 3)

    def _estimate_vram(self, path: str, quantization: str = None) -> float:
        """估算显存需求"""
        size_gb = self._get_model_size(path)
        if quantization == "awq" or quantization == "gptq":
            return size_gb * 1.2   # 量化模型 + KV Cache 开销
        else:
            return size_gb * 2.0   # FP16 模型 + KV Cache 开销
```
